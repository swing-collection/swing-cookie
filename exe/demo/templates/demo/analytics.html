Code Example

Here's how to conditionally load scripts based on consent:

<!-- This script type prevents execution until consent is given -->
<script type="x/cookie_consent" data-varname="analytics">
    // Google Analytics initialization
    gtag('config', 'UA-XXXXX-Y');
</script>

<!-- Or check programmatically -->
<script>
    if (SwingCookieConsent.isAccepted('analytics')) {
        // Initialize analytics
    }
</script>
